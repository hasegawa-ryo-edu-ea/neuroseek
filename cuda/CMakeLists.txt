cmake_minimum_required(VERSION 3.22)
project(neuroseek_cuda LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
# Orin Nano (Ampere) is SM 8.7. Override only for a verified compatible target.
set(CMAKE_CUDA_ARCHITECTURES 87 CACHE STRING "Jetson CUDA architectures")
add_library(neuroseek_cuda SHARED src/neuroseek_cuda.cu)
target_include_directories(neuroseek_cuda PUBLIC include)
add_executable(neuroseek_cuda_parity tests/parity_test.cu)
target_link_libraries(neuroseek_cuda_parity PRIVATE neuroseek_cuda)
add_executable(neuroseek_cuda_bench benchmarks/cuda_bench.cu)
target_link_libraries(neuroseek_cuda_bench PRIVATE neuroseek_cuda)
enable_testing()
add_test(NAME cuda_parity COMMAND neuroseek_cuda_parity)
set_tests_properties(cuda_parity PROPERTIES SKIP_RETURN_CODE 77)
